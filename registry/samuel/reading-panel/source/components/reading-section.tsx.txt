const GLOBE_MARKERS = [

/** Derived from the newest dated post so it never drifts. Shown in the reading-gif header. */
const LAST_UPDATED =
  formatDate(
    posts
      .map((p) => p.date)
      .filter((d): d is string => Boolean(d))
      .sort()
      .at(-1),
  ) ?? "";
